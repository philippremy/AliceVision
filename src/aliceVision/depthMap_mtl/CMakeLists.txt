# The output file for the generated gaussian constant arrays
set(GAUSS_ARRAY_OUTPUT_PATH ${CMAKE_CURRENT_BINARY_DIR}/metal/DeviceGaussianArrays.metal)

# All host sources (CXX/OBJCXX)
set(depthMap_mtl_host_sources
    cxx/CustomPatchPatternParams.cpp
    cxx/SgmDepthList.cpp
    cxx/Tile.cpp
    objcxx/DepthMapEstimator.mm
    objcxx/DepthMapUtils.mm
    objcxx/DeviceCache.mm
    objcxx/DeviceColorConversion.mm
    objcxx/DeviceCommandManager.mm
    objcxx/DeviceDepthSimilarityMap.mm
    objcxx/DeviceGaussianFilter.mm
    objcxx/DeviceManager.mm
    objcxx/DeviceMipmapImage.mm
    objcxx/DeviceMipmappedTexture.mm
    objcxx/DeviceSimilarityVolume.mm
    objcxx/DeviceUtils.mm
    objcxx/EnsureARC.mm
    objcxx/IConcurrentDeviceTask.mm
    objcxx/MemoryUtils.mm
    objcxx/NormalMapEstimator.mm
    objcxx/PatchPattern.mm
    objcxx/Refine.mm
    objcxx/Sgm.mm
    objcxx/VolumeIO.mm
)

# All device sources (i.e., Metal files containing kernel definitions)
set(depthMap_mtl_kernel_sources
    metal/DeviceColorConversion.metal
    metal/DeviceDepthSimilarityMap.metal
    metal/DeviceGaussianFilter.metal
    metal/DeviceMipmappedTexture.metal
    metal/DeviceSimilarityVolume.metal

    # Generated by custom command (see below)
    ${GAUSS_ARRAY_OUTPUT_PATH}
)

set(depthMap_mtl_private_links)

set(depthMap_mtl_public_links
    aliceVision_image
    aliceVision_mvsData
    aliceVision_mvsUtils
    aliceVision_numeric
    aliceVision_sfmData
    aliceVision_sfmDataIO
    aliceVision_system

    assimp::assimp
)

set(depthMap_mtl_system_dependencies
    "-framework Foundation"
    "-framework Metal"
    "objc"
)

# If building static libraries, the system frameworks need to be propagated to
# the final linker target (an executable)
if(BUILD_SHARED_LIBS)
  list(APPEND depthMap_mtl_private_links ${depthMap_mtl_system_dependencies})
else()
  list(APPEND depthMap_mtl_public_links ${depthMap_mtl_system_dependencies})
endif()

alicevision_add_library(aliceVision_depthMap_mtl
    USE_METAL
    SOURCES
        ${depthMap_mtl_host_sources}
        ${depthMap_mtl_kernel_sources}
    PUBLIC_LINKS
        ${depthMap_mtl_public_links}
    PRIVATE_LINKS
        ${depthMap_mtl_private_links}
)

# We require C++23 for std::mdspan
# This is okay, since we must use LLVM/Clang for ARC anyways and std::mdspan is
# supported since Apple Clang 15 (Xcode 15) and LLVM Clang 19.
# Xcode 15 is still runnable on Intel Macs
set_property(TARGET aliceVision_depthMap_mtl PROPERTY CXX_STANDARD 23)

# Enable ARC for Objective-C(++)
target_compile_options(aliceVision_depthMap_mtl PRIVATE "$<$<COMPILE_LANGUAGE:OBJC,OBJCXX>:-fobjc-arc>")

# Enable discovery of half using predefined MIN/MAX macros
target_compile_definitions(aliceVision_depthMap_mtl PUBLIC "__STDC_WANT_IEC_60559_TYPES_EXT__")

# We must pregenerate the gaussian arrays for Metal at compile time
# This will create a .metal file which is treated as a generated file and be
# included in the final binary.
# We have another header which then refers to it using "extern".
set(GAUSS_ARRAY_TOOL_PATH ${CMAKE_CURRENT_BINARY_DIR}/tools/generate_gauss_array)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/tools)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/metal)
add_custom_command(OUTPUT ${GAUSS_ARRAY_OUTPUT_PATH}
    COMMAND xcrun clang ${CMAKE_CURRENT_LIST_DIR}/tools/generate_gauss_array.c -o "${GAUSS_ARRAY_TOOL_PATH}"
    COMMAND ${GAUSS_ARRAY_TOOL_PATH} "${GAUSS_ARRAY_OUTPUT_PATH}"
    COMMENT "Generating MSL source DeviceGaussianArrays.metal"
    DEPENDS ${CMAKE_CURRENT_LIST_DIR}/Config.h
    MAIN_DEPENDENCY ${CMAKE_CURRENT_LIST_DIR}/tools/generate_gauss_array.c
    DEPENDS_EXPLICIT_ONLY
)
