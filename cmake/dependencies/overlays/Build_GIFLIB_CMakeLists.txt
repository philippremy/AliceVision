cmake_minimum_required(VERSION 3.25)

project(GIFLIB VERSION 6.1.3 LANGUAGES C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Include helpers
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

# Windows should export the functions by default, so they can be used from the
# headers
if(WIN32)
    set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()

# The GIF library
add_library(GIF
    dgif_lib.c
    egif_lib.c
    gifalloc.c
    gif_err.c
    gif_font.c
    gif_hash.c
    openbsd-reallocarray.c
    quantize.c
)

# The UTIL library
add_library(UTIL
    qprintf.c
    getarg.c
)

# The util library must link to GIF
target_link_libraries(UTIL PUBLIC GIF)

# The library output names should be lowercase
set_target_properties(GIF PROPERTIES
    OUTPUT_NAME gif
)

# The library output names should be lowercase
set_target_properties(UTIL PROPERTIES
    OUTPUT_NAME util
)

# Three GIFLIB/UTIL headers
set(GIFLIB_HEADERS
    gif_hash.h
    gif_lib.h
    gif_lib_private.h
    getarg.h
)

target_include_directories(GIF
  PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
)

target_include_directories(UTIL
  PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
)

# macOS deployment target
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "10.9"
    CACHE STRING "Minimum macOS deployment version.")
endif()

set_target_properties(GIF PROPERTIES
  POSITION_INDEPENDENT_CODE ON
  VERSION   "${PROJECT_VERSION}"
  SOVERSION "${PROJECT_VERSION_MAJOR}"
)

set_target_properties(UTIL PROPERTIES
  POSITION_INDEPENDENT_CODE ON
  VERSION   "${PROJECT_VERSION}"
  SOVERSION "${PROJECT_VERSION_MAJOR}"
)

# Install the targets
install(
  TARGETS   GIF UTIL
  EXPORT    GIFLIBTargets
  ARCHIVE   DESTINATION "${CMAKE_INSTALL_LIBDIR}"  COMPONENT library
  LIBRARY   DESTINATION "${CMAKE_INSTALL_LIBDIR}"  COMPONENT library
  RUNTIME   DESTINATION "${CMAKE_INSTALL_BINDIR}"  COMPONENT library
)

# Install the headers
install(
  FILES       ${GIFLIB_HEADERS}
  DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
  COMPONENT   headers
)

# Write a basic version file
write_basic_package_version_file(
  "${CMAKE_CURRENT_BINARY_DIR}/GIFConfigVersion.cmake"
  VERSION "${PROJECT_VERSION}"
  COMPATIBILITY SameMajorVersion
)

# Write a basic Config file from our generic template
configure_package_config_file(
  "${CMAKE_CURRENT_SOURCE_DIR}/OverlayConfig.cmake.in"
  "${CMAKE_CURRENT_BINARY_DIR}/GIFConfig.cmake"
  INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GIF"
  PATH_VARS
    CMAKE_INSTALL_LIBDIR
)

# Install version and config file
install(
  FILES
    "${CMAKE_CURRENT_BINARY_DIR}/GIFConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/GIFConfigVersion.cmake"
  DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GIF"
)

# Install targets
install(
  EXPORT      GIFLIBTargets
  NAMESPACE   "GIF::"
  DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GIF"
)
